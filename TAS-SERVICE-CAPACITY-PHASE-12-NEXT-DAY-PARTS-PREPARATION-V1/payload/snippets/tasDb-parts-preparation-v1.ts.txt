const TAS_PREPARATION_STATUSES = new Set(["Pending", "Partial", "Prepared", "Shortage"]);

const normalizeTASPreparationDay = (value: unknown) => {
  if (value instanceof Date && !Number.isNaN(value.getTime())) return value.toISOString().slice(0, 10);
  const text = String(value ?? "").trim();
  const match = text.match(/^(\d{4}-\d{2}-\d{2})/);
  if (!match) throw new Error("Valid preparation day is required");
  return match[1];
};

const addTASPreparationDay = (date: string, days: number) => {
  const match = date.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!match) throw new Error("Invalid preparation date");
  const value = new Date(Date.UTC(Number(match[1]), Number(match[2]) - 1, Number(match[3])));
  value.setUTCDate(value.getUTCDate() + days);
  return value.toISOString().slice(0, 10);
};

const normalizeTASPreparationStatus = (value: unknown) => {
  const candidate = String(value ?? "Pending").trim();
  return TAS_PREPARATION_STATUSES.has(candidate) ? candidate : "Pending";
};

const normalizeTASPreparationQuantity = (value: unknown, fallback = 0) => {
  const parsed = Number(value);
  if (!Number.isFinite(parsed)) return Math.max(0, fallback);
  return Math.round(Math.max(0, parsed) * 1000) / 1000;
};

export const getTASPartsPreparationBoard = async (input: {
  day: Date | string;
  branchId?: number | null;
}) => {
  const db = await getDb();
  if (!db) throw new Error("Database not available");

  const date = normalizeTASPreparationDay(input.day);
  const nextDate = addTASPreparationDay(date, 1);
  const branchId = Number(input.branchId ?? 0);
  const branchFilter = Number.isInteger(branchId) && branchId > 0
    ? sql`AND b.branchId = ${branchId}`
    : sql``;

  const rows = await db.execute(sql`
    SELECT
      bi.id bookingItemId,
      bi.bookingId,
      bi.itemId,
      bi.itemCode,
      bi.itemName,
      bi.itemType,
      bi.action,
      bi.unit,
      bi.quantity requiredQuantity,
      bi.preparationRequired,
      b.status bookingStatus,
      b.customerName,
      b.phone customerPhone,
      b.vehicleBrand,
      b.vehicleModel,
      b.vehicleYear,
      b.mileageKm,
      b.branchId,
      br.name branchName,
      b.maintenancePlanId,
      mp.name maintenancePlanName,
      b.maintenanceIntervalId,
      mi.label maintenanceIntervalLabel,
      mi.mileageKm maintenanceMileageKm,
      DATE_FORMAT(
        COALESCE(b.startAt, TIMESTAMP(b.preferredDate, b.preferredTime)),
        '%Y-%m-%dT%H:%i:%s'
      ) appointmentStartAt,
      prep.id preparationId,
      prep.status preparationStatus,
      prep.preparedQuantity,
      prep.notes preparationNotes,
      prep.updatedByUserId,
      prep.updatedAt preparationUpdatedAt,
      u.name preparationUpdatedByName
    FROM tas_service_booking_items bi
    JOIN tas_service_bookings b ON b.id=bi.bookingId
    LEFT JOIN tas_branches br ON br.id=b.branchId
    LEFT JOIN tas_maintenance_plans mp ON mp.id=b.maintenancePlanId
    LEFT JOIN tas_maintenance_intervals mi ON mi.id=b.maintenanceIntervalId
    LEFT JOIN tas_service_booking_item_preparation prep ON prep.bookingItemId=bi.id
    LEFT JOIN users u ON u.id=prep.updatedByUserId
    WHERE bi.preparationRequired=1
      AND b.status NOT IN ('Cancelled','NoShow','Completed')
      AND COALESCE(b.startAt, TIMESTAMP(b.preferredDate, b.preferredTime)) >= ${date + " 00:00:00"}
      AND COALESCE(b.startAt, TIMESTAMP(b.preferredDate, b.preferredTime)) < ${nextDate + " 00:00:00"}
      ${branchFilter}
    ORDER BY
      COALESCE(b.startAt, TIMESTAMP(b.preferredDate, b.preferredTime)) ASC,
      b.id ASC,
      bi.id ASC
  `);

  const raw = ((rows as any)[0] ?? []) as AnyRow[];
  const bookingIds = new Set<number>();

  const lines = raw.map((row: AnyRow) => {
    const requiredQuantity = normalizeTASPreparationQuantity(row.requiredQuantity, 0);
    const preparationStatus = normalizeTASPreparationStatus(row.preparationStatus);
    const preparedQuantity = Math.min(
      requiredQuantity,
      preparationStatus === "Prepared"
        ? requiredQuantity
        : normalizeTASPreparationQuantity(row.preparedQuantity, 0),
    );
    const remainingQuantity = Math.round(Math.max(0, requiredQuantity - preparedQuantity) * 1000) / 1000;
    const bookingId = Number(row.bookingId);
    bookingIds.add(bookingId);

    return {
      bookingItemId: Number(row.bookingItemId),
      bookingId,
      itemId: row.itemId == null ? null : Number(row.itemId),
      itemCode: row.itemCode ?? null,
      itemName: row.itemName,
      itemType: row.itemType,
      action: row.action,
      unit: row.unit ?? "unit",
      requiredQuantity,
      preparedQuantity,
      remainingQuantity,
      preparationRequired: true,
      preparationStatus,
      preparationNotes: row.preparationNotes ?? null,
      preparationUpdatedAt: row.preparationUpdatedAt ?? null,
      updatedByUserId: row.updatedByUserId == null ? null : Number(row.updatedByUserId),
      updatedByName: row.preparationUpdatedByName ?? null,
      bookingStatus: row.bookingStatus,
      customerName: row.customerName ?? null,
      customerPhone: row.customerPhone ?? null,
      vehicleBrand: row.vehicleBrand ?? null,
      vehicleModel: row.vehicleModel ?? null,
      vehicleYear: row.vehicleYear ?? null,
      mileageKm: row.mileageKm == null ? null : Number(row.mileageKm),
      branchId: row.branchId == null ? null : Number(row.branchId),
      branchName: row.branchName ?? null,
      maintenancePlanId: row.maintenancePlanId == null ? null : Number(row.maintenancePlanId),
      maintenancePlanName: row.maintenancePlanName ?? null,
      maintenanceIntervalId: row.maintenanceIntervalId == null ? null : Number(row.maintenanceIntervalId),
      maintenanceIntervalLabel: row.maintenanceIntervalLabel ?? null,
      maintenanceMileageKm: row.maintenanceMileageKm == null ? null : Number(row.maintenanceMileageKm),
      appointmentStartAt: row.appointmentStartAt ?? null,
    };
  });

  const aggregateMap = new Map<string, AnyRow>();

  for (const line of lines) {
    const key = [
      line.itemId ?? 0,
      line.itemCode ?? "",
      line.itemName,
      line.unit,
    ].join("|");

    let aggregate = aggregateMap.get(key);
    if (!aggregate) {
      aggregate = {
        key,
        itemId: line.itemId,
        itemCode: line.itemCode,
        itemName: line.itemName,
        itemType: line.itemType,
        unit: line.unit,
        requiredQuantity: 0,
        preparedQuantity: 0,
        remainingQuantity: 0,
        lineCount: 0,
        preparedLineCount: 0,
        partialLineCount: 0,
        shortageLineCount: 0,
        pendingLineCount: 0,
        bookingIds: new Set<number>(),
      };
      aggregateMap.set(key, aggregate);
    }

    aggregate.requiredQuantity += line.requiredQuantity;
    aggregate.preparedQuantity += line.preparedQuantity;
    aggregate.remainingQuantity += line.remainingQuantity;
    aggregate.lineCount += 1;
    aggregate.bookingIds.add(line.bookingId);

    if (line.preparationStatus === "Prepared") aggregate.preparedLineCount += 1;
    else if (line.preparationStatus === "Partial") aggregate.partialLineCount += 1;
    else if (line.preparationStatus === "Shortage") aggregate.shortageLineCount += 1;
    else aggregate.pendingLineCount += 1;
  }

  const aggregates = Array.from(aggregateMap.values())
    .map((row: AnyRow) => ({
      ...row,
      requiredQuantity: Math.round(row.requiredQuantity * 1000) / 1000,
      preparedQuantity: Math.round(row.preparedQuantity * 1000) / 1000,
      remainingQuantity: Math.round(row.remainingQuantity * 1000) / 1000,
      bookingCount: row.bookingIds.size,
      bookingIds: Array.from(row.bookingIds).sort((a: number, b: number) => a - b),
      ready: row.lineCount > 0 && row.preparedLineCount === row.lineCount,
    }))
    .sort((a: AnyRow, b: AnyRow) => {
      if (b.shortageLineCount !== a.shortageLineCount) return b.shortageLineCount - a.shortageLineCount;
      if (b.remainingQuantity !== a.remainingQuantity) return b.remainingQuantity - a.remainingQuantity;
      return String(a.itemName).localeCompare(String(b.itemName));
    });

  const readyLineCount = lines.filter((line: AnyRow) => line.preparationStatus === "Prepared").length;
  const shortageLineCount = lines.filter((line: AnyRow) => line.preparationStatus === "Shortage").length;
  const partialLineCount = lines.filter((line: AnyRow) => line.preparationStatus === "Partial").length;
  const pendingLineCount = lines.filter((line: AnyRow) => line.preparationStatus === "Pending").length;
  const readinessPct = lines.length > 0 ? Math.round((readyLineCount / lines.length) * 1000) / 10 : 0;

  return {
    preparationVersion: 1,
    date,
    branchId: Number.isInteger(branchId) && branchId > 0 ? branchId : null,
    lines,
    aggregates,
    summary: {
      bookingCount: bookingIds.size,
      distinctItemCount: aggregates.length,
      lineCount: lines.length,
      readyLineCount,
      partialLineCount,
      shortageLineCount,
      pendingLineCount,
      readinessPct,
    },
  };
};

export const updateTASBookingItemPreparation = async (input: {
  bookingItemId: number;
  status: string;
  preparedQuantity?: number | null;
  notes?: string | null;
  updatedByUserId?: number | null;
}) => {
  const db = await getDb();
  if (!db) throw new Error("Database not available");

  const bookingItemId = Number(input.bookingItemId ?? 0);
  if (!Number.isInteger(bookingItemId) || bookingItemId <= 0) {
    throw new Error("Valid bookingItemId is required");
  }

  const status = String(input.status ?? "").trim();
  if (!TAS_PREPARATION_STATUSES.has(status)) throw new Error("Invalid preparation status");

  const notes = String(input.notes ?? "").trim() || null;
  const updatedBy = Number(input.updatedByUserId ?? 0);
  const updatedByUserId = Number.isInteger(updatedBy) && updatedBy > 0 ? updatedBy : null;

  return db.transaction(async (tx) => {
    const rows = await tx.execute(sql`
      SELECT
        bi.id bookingItemId,
        bi.bookingId,
        bi.quantity requiredQuantity,
        bi.preparationRequired,
        b.status bookingStatus
      FROM tas_service_booking_items bi
      JOIN tas_service_bookings b ON b.id=bi.bookingId
      WHERE bi.id=${bookingItemId}
      LIMIT 1
      FOR UPDATE
    `);
    const row = ((rows as any)[0] ?? [])[0] as AnyRow | undefined;
    if (!row) throw new Error("Booking preparation item not found");
    if (Number(row.preparationRequired) !== 1) throw new Error("This booking item does not require preparation");
    if (["Cancelled", "NoShow", "Completed"].includes(String(row.bookingStatus ?? ""))) {
      throw new Error("Preparation cannot be updated for a closed booking");
    }

    const requiredQuantity = normalizeTASPreparationQuantity(row.requiredQuantity, 0);
    let preparedQuantity = normalizeTASPreparationQuantity(input.preparedQuantity, 0);

    if (status === "Pending") preparedQuantity = 0;
    if (status === "Prepared") preparedQuantity = requiredQuantity;

    if (preparedQuantity > requiredQuantity) {
      throw new Error("Prepared quantity cannot exceed the required quantity");
    }
    if (status === "Partial" && !(preparedQuantity > 0 && preparedQuantity < requiredQuantity)) {
      throw new Error("Partial status requires a prepared quantity between zero and the required quantity");
    }
    if (status === "Shortage") {
      if (preparedQuantity >= requiredQuantity) {
        throw new Error("Shortage status requires a remaining quantity");
      }
      if (!notes) throw new Error("Shortage reason is required");
    }

    await tx.execute(sql`
      INSERT INTO tas_service_booking_item_preparation (
        bookingItemId,
        bookingId,
        status,
        preparedQuantity,
        notes,
        updatedByUserId,
        createdAt,
        updatedAt
      ) VALUES (
        ${bookingItemId},
        ${Number(row.bookingId)},
        ${status},
        ${preparedQuantity},
        ${notes},
        ${updatedByUserId},
        CURRENT_TIMESTAMP,
        CURRENT_TIMESTAMP
      )
      ON DUPLICATE KEY UPDATE
        bookingId=VALUES(bookingId),
        status=VALUES(status),
        preparedQuantity=VALUES(preparedQuantity),
        notes=VALUES(notes),
        updatedByUserId=VALUES(updatedByUserId),
        updatedAt=CURRENT_TIMESTAMP
    `);

    return {
      success: true,
      bookingItemId,
      bookingId: Number(row.bookingId),
      status,
      requiredQuantity,
      preparedQuantity,
      remainingQuantity: Math.round(Math.max(0, requiredQuantity - preparedQuantity) * 1000) / 1000,
      notes,
    };
  });
};
